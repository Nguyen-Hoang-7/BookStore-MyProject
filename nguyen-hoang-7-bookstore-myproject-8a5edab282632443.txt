# React + Vite

This template provides a minimal setup to get React working in Vite with HMR and some ESLint rules.

Currently, two official plugins are available:

## React Compiler

The React Compiler is not enabled on this template because of its impact on dev & build performances. To add it, see [this documentation](https://react.dev/learn/react-compiler/installation).

If you are developing a production application, we recommend using TypeScript with type-aware lint rules enabled. // src/components/admin/BookFormModal.jsx
import { useEffect, useState } from "react";
import { categoryApi } from "../../api/categoryApi";
import { authorApi } from "../../api/authorApi";
import { publisherApi } from "../../api/publisherApi";
import { getErrorMessage } from "../../api/axiosClient";

const emptyForm = {
  name: "",
  description: "",
  price: 0,
  stock: 0,
  imageUrls: [""],
  categoryId: "",
  bookDetail: { isbn: "", pages: "", language: "Tiếng Việt", publisherId: "", authorIds: [] },
};

export default function BookFormModal({ open, book, onClose, onSubmit }) {
  const [form, setForm] = useState(emptyForm);
  const [categories, setCategories] = useState([]);
  const [authors, setAuthors] = useState([]);
  const [publishers, setPublishers] = useState([]);
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    if (!open) return;
    categoryApi.getAll("Book").then((res) => setCategories(res.data));
    authorApi.getAll().then((res) => setAuthors(res.data));
    publisherApi.getAll().then((res) => setPublishers(res.data));

    if (book) {
      setForm({
        name: book.name,
        description: book.description || "",
        price: book.price,
        stock: book.stock,
        imageUrls: book.imageUrls?.length ? book.imageUrls : [""],
        categoryId: book.categoryId,
        bookDetail: {
          isbn: book.bookDetail?.isbn || "",
          pages: book.bookDetail?.pages || "",
          language: book.bookDetail?.language || "Tiếng Việt",
          publisherId: book.bookDetail?.publisher?.id || "",
          authorIds: book.bookDetail?.authors?.map((a) => a.id) || [],
        },
      });
    } else {
      setForm(emptyForm);
    }
    setError("");
  }, [open, book]);

  if (!open) return null;

  const handleChange = (e) => {
    const { name, value } = e.target;
    setForm((prev) => ({ ...prev, [name]: value }));
  };

  const handleBookDetailChange = (e) => {
    const { name, value } = e.target;
    setForm((prev) => ({ ...prev, bookDetail: { ...prev.bookDetail, [name]: value } }));
  };

  const handleImageChange = (value) => {
    setForm((prev) => ({ ...prev, imageUrls: [value] }));
  };

  const toggleAuthor = (authorId) => {
    setForm((prev) => {
      const current = prev.bookDetail.authorIds;
      const next = current.includes(authorId)
        ? current.filter((id) => id !== authorId)
        : [...current, authorId];
      return { ...prev, bookDetail: { ...prev.bookDetail, authorIds: next } };
    });
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError("");
    setLoading(true);

    try {
      await onSubmit({
        ...form,
        imageUrls: form.imageUrls.filter(Boolean),

// src/context/AuthContext.jsx
import { createContext, useContext, useState } from "react";
import { authApi } from "../api/authApi";

const AuthContext = createContext();
export const useAuth = () => useContext(AuthContext);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(() => {
    const saved = localStorage.getItem("user");
    return saved ? JSON.parse(saved) : null;
  });

  const saveSession = (data) => {
    localStorage.setItem("token", data.token);
    localStorage.setItem("user", JSON.stringify(data.user));
    setUser(data.user);
    return data.user;
  };

  const login = async (email, password) => {
    const res = await authApi.login({ email, password });
    return saveSession(res.data);
  };

  const register = async (payload) => {
    const res = await authApi.register(payload);
    return saveSession(res.data);
  };

  const logout = () => {
    localStorage.removeItem("token");
    localStorage.removeItem("user");
    setUser(null);
  };

  const isAdmin = user?.role === "Admin";

  return (
    <AuthContext.Provider value={{ user, isAdmin, login, register, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

// src/context/ThemeContext.jsx
import { createContext, useState, useContext, useEffect } from "react";

const ThemeContext = createContext(undefined);

export const ThemeProvider = ({ children }) => {
  const [theme, setTheme] = useState("light");
  const [isInitialized, setIsInitialized] = useState(false);

  useEffect(() => {
    // Chỉ chạy một lần khi component mount, đọc theme đã lưu từ lần trước
    const savedTheme = localStorage.getItem("theme");
    const initialTheme = savedTheme || "light"; // Mặc định là light theme

    setTheme(initialTheme);
    setIsInitialized(true);
  }, []);

  useEffect(() => {
    if (isInitialized) {
      localStorage.setItem("theme", theme);
      if (theme === "dark") {
        document.documentElement.classList.add("dark");
        document.documentElement.setAttribute("data-color-scheme", "dark");
      } else {
        document.documentElement.classList.remove("dark");
        document.documentElement.setAttribute("data-color-scheme", "light");
      }
    }
  }, [theme, isInitialized]);

  const toggleTheme = () => {
    setTheme((prevTheme) => (prevTheme === "light" ? "dark" : "light"));
  };

  return (
    <ThemeContext.Provider value={{ theme, toggleTheme }}>
      {children}
    </ThemeContext.Provider>
  );
};

// src/pages/admin/AdminBooks.jsx
import { useEffect, useState } from "react";
import { useNavigate } from "react-router-dom";
import { bookApi } from "../../api/bookApi";
import { getErrorMessage } from "../../api/axiosClient";
import BookTable from "../../components/admin/BookTable";
import ConfirmDialog from "../../components/common/ConfirmDialog";
import Pagination from "../../components/common/Pagination";

export default function AdminBooks() {
  const navigate = useNavigate();
  const [books, setBooks] = useState([]);
  const [total, setTotal] = useState(0);
  const [page, setPage] = useState(1);
  const [search, setSearch] = useState("");
  const [loading, setLoading] = useState(false);

  const [deletingBook, setDeletingBook] = useState(null);
  const [deleting, setDeleting] = useState(false);

  const pageSize = 10;
  const totalPages = Math.ceil(total / pageSize);

  const fetchBooks = async () => {
    setLoading(true);
    try {
      const res = await bookApi.getAll({ q: search, page, pageSize });
      setBooks(res.data.items);
      setTotal(res.data.total);
    } catch (err) {
      alert(getErrorMessage(err));
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchBooks();
  }, [page]);

  const handleSearchSubmit = (e) => {
    e.preventDefault();
    setPage(1);
    fetchBooks();
  };

  const confirmDelete = async () => {
    setDeleting(true);
    try {
      await bookApi.remove(deletingBook.id);
      setDeletingBook(null);
      fetchBooks();
    } catch (err) {
      alert(getErrorMessage(err));
    } finally {
      setDeleting(false);
    }
  };

  return (
    <div>
      <div className="flex items-center justify-between mb-6">
        <h2 className="text-xl font-bold text-gray-900 dark:text-white">Quản lý sách</h2>
        <button
          onClick={() => navigate("/admin/books/new")}
          className="px-4 py-2 text-sm font-medium rounded-lg bg-indigo-600 text-white hover:bg-indigo-700"
        >
          + Thêm sách
        </button>
      </div>

      <form onSubmit={handleSearchSubmit} className="mb-4">
        <input
          type="text"
          placeholder="Tìm theo tên sách..."
          value={search}
          onChange={(e) => setSearch(e.target.value)}
          className="w-full max-w-sm rounded-lg border border-slate-200 dark:border-gray-700
                     dark:bg-gray-900 dark:text-white px-4 py-2 text-sm
                     focus:outline-none focus:ring-2 focus:ring-indigo-500"
        />
      </form>

      {loading ? (
        <p className="text-sm text-gray-400 text-center py-10">Đang tải...</p>
      ) : (
        <>
          <BookTable
            books={books}
            onEdit={(book) => navigate(`/admin/books/${book.id}/edit`)}
            onDelete={setDeletingBook}
          />
          <Pagination page={page} totalPages={totalPages} onChange={setPage} />
        </>
      )}

      <ConfirmDialog
        open={!!deletingBook}
        title="Xóa sách"
        message={`Bạn có chắc muốn xóa "${deletingBook?.name}"? Hành động này không thể hoàn tác.`}
        onConfirm={confirmDelete}
        onCancel={() => setDeletingBook(null)}
        loading={deleting}
      />
    </div>
  );
}

// src/pages/admin/BookFormPage.jsx
import { useEffect, useState } from "react";
import { useNavigate, useParams, Link } from "react-router-dom";
import { bookApi } from "../../api/bookApi";
import { categoryApi } from "../../api/categoryApi";
import { authorApi } from "../../api/authorApi";
import { publisherApi } from "../../api/publisherApi";
import { uploadApi } from "../../api/uploadApi";
import { getErrorMessage } from "../../api/axiosClient";

const FILE_BASE_URL = import.meta.env.VITE_FILE_BASE_URL;
const MAX_IMAGES = 10;

const emptyForm = {
  name: "",
  description: "",
  price: "",
  stock: "",
  imageUrls: [],
  categoryId: "",
  bookDetail: { isbn: "", pages: "", language: "Tiếng Việt", publisherId: "", authorIds: [] },
};

export default function BookFormPage() {
  const { id } = useParams();
  const isEdit = Boolean(id);
  const navigate = useNavigate();

  const [form, setForm] = useState(emptyForm);
  const [categories, setCategories] = useState([]);
  const [authors, setAuthors] = useState([]);
  const [publishers, setPublishers] = useState([]);
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);
  const [uploading, setUploading] = useState(false);
  const [initializing, setInitializing] = useState(isEdit);

  useEffect(() => {
    categoryApi.getAll("Book").then((res) => setCategories(res.data));
    authorApi.getAll().then((res) => setAuthors(res.data));
    publisherApi.getAll().then((res) => setPublishers(res.data));

    if (isEdit) {
      bookApi.getById(id).then((res) => {
        const book = res.data;
        setForm({
          name: book.name,
          description: book.description || "",
          price: book.price,
          stock: book.stock,
          imageUrls: book.imageUrls || [],
          categoryId: book.categoryId,
          bookDetail: {
            isbn: book.bookDetail?.isbn || "",
            pages: book.bookDetail?.pages || "",
            language: book.bookDetail?.language || "Tiếng Việt",
            publisherId: book.bookDetail?.publisher?.id || "",
            authorIds: book.bookDetail?.authors?.map((a) => a.id) || [],
          },
        });
        setInitializing(false);
      });
    }
  }, [id, isEdit]);

  const handleChange = (e) => {
    const { name, value } = e.target;
    setForm((prev) => ({ ...prev, [name]: value }));
  };

  const handleBookDetailChange = (e) => {
    const { name, value } = e.target;
    setForm((prev) => ({ ...prev, bookDetail: { ...prev.bookDetail, [name]: value } }));
  };

  // ===== Xử lý nhiều ảnh =====
  const handleFilesSelect = async (e) => {
    const files = Array.from(e.target.files || []);
    if (files.length === 0) return;

    const remainingSlots = MAX_IMAGES - form.imageUrls.length;
    if (remainingSlots <= 0) {
      setError(`Chỉ được tối đa ${MAX_IMAGES} ảnh.`);
      e.target.value = "";
      return;
    }

    const filesToUpload = files.slice(0, remainingSlots);

    setError("");
    setUploading(true);

    try {
      const res = await uploadApi.uploadImages(filesToUpload);
      const newUrls = res.data.uploaded.map((item) => item.url);

      setForm((prev) => ({
        ...prev,
        imageUrls: [...prev.imageUrls, ...newUrls],
      }));

      if (res.data.errors.length > 0) {
        setError(res.data.errors.join(" — "));
      }
    } catch (err) {
      setError(getErrorMessage(err));
    } finally {
      setUploading(false);
      e.target.value = "";
    }
  };

  const handleRemoveImage = (index) => {
    setForm((prev) => ({
      ...prev,
      imageUrls: prev.imageUrls.filter((_, i) => i !== index),
    }));
  };

  const handleSetCover = (index) => {
    setForm((prev) => {
      const urls = [...prev.imageUrls];
      const [selected] = urls.splice(index, 1);
      return { ...prev, imageUrls: [selected, ...urls] };
    });
  };

  const getImageSrc = (url) =>
    url.startsWith("http") ? url : `${FILE_BASE_URL}${url}`;

  const toggleAuthor = (authorId) => {
    setForm((prev) => {
      const current = prev.bookDetail.authorIds;
      const next = current.includes(authorId)
        ? current.filter((aid) => aid !== authorId)
        : [...current, authorId];
      return { ...prev, bookDetail: { ...prev.bookDetail, authorIds: next } };
    });
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError("");
    setLoading(true);

    const payload = {
      ...form,
      price: Number(form.price),
      stock: Number(form.stock),
      categoryId: Number(form.categoryId),
      bookDetail: {
        ...form.bookDetail,
        pages: form.bookDetail.pages ? Number(form.bookDetail.pages) : null,
        publisherId: form.bookDetail.publisherId ? Number(form.bookDetail.publisherId) : null,
      },
    };

    try {
      if (isEdit) {
        await bookApi.update(id, payload);
      } else {
        await bookApi.create(payload);
      }
      navigate("/admin/books");
    } catch (err) {
      setError(getErrorMessage(err));
      setLoading(false);
    }
  };

  const inputClass =
    "w-full rounded-lg border border-slate-200 dark:border-gray-700 dark:bg-gray-800 dark:text-white " +
    "px-3.5 py-2.5 text-sm placeholder:text-slate-400 focus:outline-none focus:ring-2 focus:ring-indigo-500 focus:border-transparent transition";

  const labelClass = "block text-left text-sm font-medium text-gray-700 dark:text-gray-300 mb-1.5";

  const cardClass = "bg-white dark:bg-gray-800 rounded-2xl border border-slate-200 dark:border-gray-700 p-6";

  if (initializing) {
    return <p className="text-sm text-gray-400 text-center py-20">Đang tải dữ liệu...</p>;
  }

  return (
    <div>
      {/* ===== Breadcrumb + Tiêu đề ===== */}
      <div className="flex items-center justify-between mb-6">
        <div>
          <h2 className="text-xl font-bold text-gray-900 dark:text-white">
            {isEdit ? "Sửa sách" : "Thêm sách mới"}
          </h2>
          <div className="flex items-center gap-1.5 text-sm text-gray-400 mt-1">
            <Link to="/admin/dashboard" className="hover:text-indigo-600">Quản trị</Link>
            <span>›</span>
            <Link to="/admin/books" className="hover:text-indigo-600">Sách</Link>
            <span>›</span>
            <span className="text-gray-600 dark:text-gray-300">{isEdit ? "Sửa" : "Thêm mới"}</span>
          </div>
        </div>

        <Link
          to="/admin/books"
          className="px-4 py-2 text-sm font-medium rounded-lg border border-slate-200 dark:border-gray-700
                     text-gray-600 dark:text-gray-300 hover:bg-gray-50 dark:hover:bg-gray-700 transition"
        >
          ← Quay lại danh sách
        </Link>
      </div>

      {error && (
        <div className="mb-5 rounded-lg bg-red-50 dark:bg-red-900/20 px-4 py-3 text-sm text-red-600 dark:text-red-400">
          {error}
        </div>
      )}

      <form onSubmit={handleSubmit} className="grid grid-cols-1 lg:grid-cols-3 gap-6">

        {/* ===== Cột trái: thông tin chính ===== */}
        <div className="lg:col-span-2 space-y-6">

          <div className={cardClass}>
            <h3 className="font-semibold text-gray-900 dark:text-white mb-4">Thông tin cơ bản</h3>
            <div className="space-y-4">
              <div>
                <label className={labelClass}>Tên sách</label>
                <input name="name" required value={form.name} onChange={handleChange} className={inputClass} placeholder="VD: Nhà giả kim" />
              </div>
              <div>
                <label className={labelClass}>Mô tả</label>
                <textarea name="description" rows={5} value={form.description} onChange={handleChange} className={inputClass} />
              </div>
            </div>
          </div>

          <div className={cardClass}>
            <h3 className="font-semibold text-gray-900 dark:text-white mb-4">Chi tiết sách</h3>

namespace BookStoreApi.Controllers;

public abstract class BaseApiController : ControllerBase
{
    protected IActionResult ToErrorResponse<T>(ServiceResult<T> result)
    {
        var body = new { message = result.ErrorMessage };
        return result.ErrorType switch
        {
            ServiceErrorType.Conflict => Conflict(body),
            ServiceErrorType.Unauthorized => Unauthorized(body),
            ServiceErrorType.Forbidden => StatusCode(403, body),
            ServiceErrorType.NotFound => NotFound(body),
            _ => BadRequest(body)
        };
    }
}

using Microsoft.EntityFrameworkCore;
using BookStoreApi.Models;
using System.Text.Json;
using Microsoft.EntityFrameworkCore.ChangeTracking;

namespace BookStoreApi.Data;

public class AppDbContext(DbContextOptions<AppDbContext> options) : DbContext(options)
{
    public DbSet<User> Users => Set<User>();
    public DbSet<Category> Categories => Set<Category>();
    public DbSet<Product> Products => Set<Product>();
    public DbSet<BookDetail> BookDetails => Set<BookDetail>();
    public DbSet<Publisher> Publishers => Set<Publisher>();
    public DbSet<Author> Authors => Set<Author>();

    protected override void OnModelCreating(ModelBuilder b)
    {
        var imageUrlsComparer = new ValueComparer<List<string>>(
            (a, b) => (a ?? new()).SequenceEqual(b ?? new()),
            v => v.Aggregate(0, (hash, s) => HashCode.Combine(hash, s.GetHashCode())),
            v => v.ToList()
        );

        b.Entity<User>(e =>
        {
            e.Property(x => x.Email).HasMaxLength(256);
            e.HasIndex(x => x.Email).IsUnique();       // không cho trùng email

            e.Property(x => x.FullName).HasMaxLength(150);
            e.Property(x => x.PasswordHash).HasMaxLength(255);
            e.Property(x => x.PhoneNumber).HasMaxLength(20);
            e.Property(x => x.Address).HasMaxLength(300);
            e.Property(x => x.Role).HasMaxLength(20);
        });

        // ==========================================
        // 2. BẢNG CATEGORIES (Danh mục)
        // ==========================================
        b.Entity<Category>(e =>
        {
            e.ToTable("Categories");
            e.HasKey(x => x.Id);

            e.Property(x => x.Name).HasMaxLength(150).IsRequired();
            e.Property(x => x.Slug).HasMaxLength(160).IsRequired();
            e.HasIndex(x => x.Slug).IsUnique(); // Slug không trùng phục vụ SEO

            e.Property(x => x.Description).HasMaxLength(500);

            // Lưu Enum CategoryType ("Book", "Other") dưới dạng chuỗi
            e.Property(x => x.Type)
             .HasConversion<string>()
             .HasMaxLength(20)
             .IsRequired();
        });

        // ==========================================
        // 3. BẢNG PRODUCTS (Sản phẩm chung)
        // ==========================================
        b.Entity<Product>(e =>
        {
            e.ToTable("Products");
            e.HasKey(x => x.Id);

            e.Property(x => x.Name).HasMaxLength(250).IsRequired();
            e.Property(x => x.Slug).HasMaxLength(260).IsRequired();
            e.HasIndex(x => x.Slug).IsUnique();

            e.Property(x => x.Price).HasColumnType("decimal(18,2)").IsRequired();
            
            // Cho phép văn bản mô tả dài vô hạn (nvarchar(max))
            e.Property(x => x.Description).HasColumnType("nvarchar(max)");

            // Chuyển List<string> (nhiều ảnh) sang dạng chuỗi JSON lưu trong SQL Server
            e.Property(x => x.ImageUrls)
             .HasConversion(
                 v => JsonSerializer.Serialize(v, (JsonSerializerOptions?)null),
                 v => JsonSerializer.Deserialize<List<string>>(v, (JsonSerializerOptions?)null) ?? new List<string>()
             )
             .Metadata.SetValueComparer(imageUrlsComparer);

            // Quan hệ 1-Nhiều với Category (1 Category có nhiều Product)
            e.HasOne(x => x.Category)
             .WithMany(c => c.Products)
             .HasForeignKey(x => x.CategoryId)
             .OnDelete(DeleteBehavior.Restrict); // Tránh xóa nhầm danh mục làm xóa sạch sản phẩm
        });

        // ==========================================
        // 4. BẢNG PUBLISHERS (Nhà xuất bản)
        // ==========================================
        b.Entity<Publisher>(e =>
        {
            e.ToTable("Publishers");
            e.HasKey(x => x.Id);

            e.Property(x => x.Name).HasMaxLength(200).IsRequired();
            e.Property(x => x.Address).HasMaxLength(300);
            e.Property(x => x.PhoneNumber).HasMaxLength(20);
        });

        // ==========================================
        // 5. BẢNG AUTHORS (Tác giả)
        // ==========================================
        b.Entity<Author>(e =>
        {
            e.ToTable("Authors");
            e.HasKey(x => x.Id);

            e.Property(x => x.Name).HasMaxLength(150).IsRequired();
            e.Property(x => x.Biography).HasColumnType("nvarchar(max)");
            e.Property(x => x.AvatarUrl).HasMaxLength(500);
        });

        // ==========================================
        // 6. BẢNG BOOKDETAILS (Chi tiết sách)
        // ==========================================
        b.Entity<BookDetail>(e =>
        {
            e.ToTable("BookDetails");

            // ProductId vừa là Primary Key vừa là Foreign Key
            e.HasKey(x => x.ProductId);

            e.Property(x => x.Isbn).HasMaxLength(50);
            e.Property(x => x.Language).HasMaxLength(50);

            // Quan hệ 1 - 1 với Product
            e.HasOne(x => x.Product)
             .WithOne(p => p.BookDetail)
             .HasForeignKey<BookDetail>(x => x.ProductId)
             .OnDelete(DeleteBehavior.Cascade); // Xóa Product thì xóa luôn BookDetail

            // Quan hệ N - 1 với Publisher (Nhiều sách thuộc 1 NXB)
            e.HasOne(x => x.Publisher)
             .WithMany(p => p.BookDetails)
             .HasForeignKey(x => x.PublisherId)
             .OnDelete(DeleteBehavior.SetNull); // Xóa NXB thì để null chứ không xóa sách

            // Quan hệ Nhiều - Nhiều (N - N) giữa BookDetail và Author
            // Tự động tạo bảng trung gian "BookDetailAuthors"
            e.HasMany(bd => bd.Authors)
             .WithMany(a => a.BookDetails)
             .UsingEntity(j => j.ToTable("BookDetailAuthors"));
        });
    }
}

using BookStoreApi.Models;
using Microsoft.EntityFrameworkCore;

namespace BookStoreApi.Data;

public static class DbSeeder
{
    public static async Task SeedAdminAsync(IServiceProvider services)
    {
        var db = services.GetRequiredService<AppDbContext>();
        var config = services.GetRequiredService<IConfiguration>();

        await db.Database.MigrateAsync();

        var email = config["SeedAdmin:Email"];
        var password = config["SeedAdmin:Password"];
        if (string.IsNullOrEmpty(email) || string.IsNullOrEmpty(password))
            return;

        email = email.Trim().ToLowerInvariant();
        if (await db.Users.AnyAsync(u => u.Email == email))
            return;   // đã có Admin này rồi thì không tạo lại

        db.Users.Add(new User
        {
            FullName = "Quản trị viên",
            Email = email,
            PasswordHash = BCrypt.Net.BCrypt.HashPassword(password),
            Role = "Admin",
            IsActive = true
        });

        await db.SaveChangesAsync();
    }
}

using BookStoreApi.Models;

namespace BookStoreApi.DTOs.Catalog;

public static class CatalogMappingExtensions
{
    public static AuthorDTO ToDto(this Author a) => new()
    {
        Id = a.Id,
        Name = a.Name,
        Biography = a.Biography,
        AvatarUrl = a.AvatarUrl
    };

    public static PublisherDTO ToDto(this Publisher p) => new()
    {
        Id = p.Id,
        Name = p.Name,
        Address = p.Address,
        PhoneNumber = p.PhoneNumber
    };

    public static CategoryDTO ToDto(this Category c) => new()
    {
        Id = c.Id,
        Name = c.Name,
        Slug = c.Slug,
        Description = c.Description,
        Type = c.Type,
        ProductCount = c.Products?.Count ?? 0
    };

    public static ProductDTO ToDto(this Product p) => new()
    {
        Id = p.Id,
        Name = p.Name,
        Slug = p.Slug,
        Description = p.Description,
        Price = p.Price,
        Stock = p.Stock,
        ImageUrls = p.ImageUrls,
        CategoryId = p.CategoryId,
        CategoryName = p.Category?.Name ?? string.Empty,
        BookDetail = p.BookDetail is null ? null : new BookDetailDTO
        {
            Isbn = p.BookDetail.Isbn,
            Pages = p.BookDetail.Pages,
            Language = p.BookDetail.Language,
            PublishedDate = p.BookDetail.PublishedDate,
            Publisher = p.BookDetail.Publisher?.ToDto(),
            Authors = p.BookDetail.Authors.Select(a => a.ToDto()).ToList()
        }
    };

    // Dùng chung cho cả Category lẫn Product vì cùng có Slug
    public static string ToSlug(this string input)
    {
        var normalized = input.Trim().ToLowerInvariant()
            .Normalize(System.Text.NormalizationForm.FormD);

        var sb = new System.Text.StringBuilder();
        foreach (var ch in normalized)
        {
            var cat = System.Globalization.CharUnicodeInfo.GetUnicodeCategory(ch);
            if (cat != System.Globalization.UnicodeCategory.NonSpacingMark)
                sb.Append(ch);
        }

        var slug = sb.ToString().Normalize(System.Text.NormalizationForm.FormC);
        slug = System.Text.RegularExpressions.Regex.Replace(slug, @"[^a-z0-9\s-]", "");
        slug = System.Text.RegularExpressions.Regex.Replace(slug, @"\s+", "-").Trim('-');
        return slug;
    }
}

using System.Globalization;
using System.Text;
using System.Text.RegularExpressions;

namespace BookStoreApi.Helpers;

public static class SlugHelper
{
    public static string GenerateSlug(string title)
    {
        if (string.IsNullOrWhiteSpace(title)) return string.Empty;

        var normalizedString = title.Trim().ToLowerInvariant().Replace("đ", "d").Normalize(NormalizationForm.FormD);
        var stringBuilder = new StringBuilder();

        foreach (var c in normalizedString)
        {
            var unicodeCategory = CharUnicodeInfo.GetUnicodeCategory(c);
            if (unicodeCategory != UnicodeCategory.NonSpacingMark)
            {
                stringBuilder.Append(c);
            }
        }

        var result = stringBuilder.ToString().Normalize(NormalizationForm.FormC);
        result = Regex.Replace(result, @"[^a-z0-9\s-]", "");
        result = Regex.Replace(result, @"\s+", "-").Trim('-');

        return result;
    }
}

// Services/TokenService.cs
using System.IdentityModel.Tokens.Jwt;
using System.Security.Claims;
using System.Text;
using BookStoreApi.DTOs.Auth;
using BookStoreApi.DTOs.User;
using BookStoreApi.Models;
using Microsoft.IdentityModel.Tokens;

namespace BookStoreApi.Services;

public class TokenService : ITokenService
{
    private readonly IConfiguration _config;

    public TokenService(IConfiguration config) => _config = config;

    public AuthResponseDTO CreateToken(User user)
    {
        var jwtKey = _config["Jwt:Key"]
            ?? throw new InvalidOperationException("Chưa cấu hình Jwt:Key");

        var claims = new[]
        {
            new Claim(ClaimTypes.NameIdentifier, user.Id.ToString()),
            new Claim(ClaimTypes.Email, user.Email),
            new Claim(ClaimTypes.Name, user.FullName),
            new Claim(ClaimTypes.Role, user.Role)
        };

        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(jwtKey));
        var expires = DateTime.UtcNow.AddMinutes(double.Parse(_config["Jwt:DurationInMinutes"] ?? "20"));

        var token = new JwtSecurityToken(
            issuer: _config["Jwt:Issuer"],
            audience: _config["Jwt:Audience"],
            claims: claims,
            expires: expires,
            signingCredentials: new SigningCredentials(key, SecurityAlgorithms.HmacSha256));

        return new AuthResponseDTO
        {
            Token = new JwtSecurityTokenHandler().WriteToken(token),
            ExpiresAt = expires,
            User = user.ToDTO()
        };
    }
}

// Services/Admin/AdminUserService.cs
using BookStoreApi.Data;
using BookStoreApi.DTOs.Admin;
using Microsoft.EntityFrameworkCore;

namespace BookStoreApi.Services.Admin;

public class AdminUserService : IAdminUserService
{
    private readonly AppDbContext _db;

    public AdminUserService(AppDbContext db) => _db = db;

    public async Task<ServiceResult<object>> GetAllAsync(string? q, int page, int pageSize)
    {
        var query = _db.Users.AsNoTracking().AsQueryable();

        if (!string.IsNullOrWhiteSpace(q))
            query = query.Where(u => u.Email.Contains(q) || u.FullName.Contains(q));

        var total = await query.CountAsync();
        var items = await query
            .OrderByDescending(u => u.CreatedAt)
            .Skip((page - 1) * pageSize)
            .Take(pageSize)
            .Select(u => u.ToListItemDTO())
            .ToListAsync();

        return ServiceResult<object>.Ok(new { total, page, pageSize, items });
    }

    public async Task<ServiceResult<bool>> LockAsync(int targetUserId, int currentAdminId)
    {
        if (targetUserId == currentAdminId)
            return ServiceResult<bool>.Fail("Không thể tự khóa tài khoản của mình.");

        var user = await _db.Users.FindAsync(targetUserId);
        if (user is null) return ServiceResult<bool>.Fail("Không tìm thấy user.", ServiceErrorType.NotFound);

        user.IsActive = false;
        user.UpdatedAt = DateTime.UtcNow;
        await _db.SaveChangesAsync();
        return ServiceResult<bool>.Ok(true);
    }

    public async Task<ServiceResult<bool>> UnlockAsync(int targetUserId)
    {
        var user = await _db.Users.FindAsync(targetUserId);
        if (user is null) return ServiceResult<bool>.Fail("Không tìm thấy user.", ServiceErrorType.NotFound);

        user.IsActive = true;
        user.UpdatedAt = DateTime.UtcNow;
        await _db.SaveChangesAsync();
        return ServiceResult<bool>.Ok(true);
    }

    public async Task<ServiceResult<bool>> SetRoleAsync(int targetUserId, int currentAdminId, AdminUpdateRoleDTO dto)
    {
        if (dto.Role is not ("Admin" or "Customer"))
            return ServiceResult<bool>.Fail("Vai trò không hợp lệ.");

        if (targetUserId == currentAdminId)
            return ServiceResult<bool>.Fail("Không thể tự đổi vai trò của mình.");

        var user = await _db.Users.FindAsync(targetUserId);
        if (user is null) return ServiceResult<bool>.Fail("Không tìm thấy user.", ServiceErrorType.NotFound);

        user.Role = dto.Role;
        user.UpdatedAt = DateTime.UtcNow;
        await _db.SaveChangesAsync();
        return ServiceResult<bool>.Ok(true);
    }
}

    private readonly AppDbContext _db;
    private readonly ITokenService _tokenService;

    public AuthService(AppDbContext db, ITokenService tokenService)
    {
        _db = db;
        _tokenService = tokenService;
    }

    public async Task<ServiceResult<AuthResponseDTO>> RegisterAsync(RegisterDTO dto)
    {
        var email = dto.Email.Trim().ToLowerInvariant();

        if (await _db.Users.AnyAsync(u => u.Email == email))
            return ServiceResult<AuthResponseDTO>.Fail("Email đã được sử dụng.", ServiceErrorType.Conflict);

        var user = new User
        {
            FullName = dto.FullName.Trim(),
            Email = email,
            PasswordHash = BCrypt.Net.BCrypt.HashPassword(dto.Password),
            PhoneNumber = dto.PhoneNumber ?? string.Empty,
            Role = "Customer"
        };

        _db.Users.Add(user);
        await _db.SaveChangesAsync();

        return ServiceResult<AuthResponseDTO>.Ok(_tokenService.CreateToken(user));
    }

    public async Task<ServiceResult<AuthResponseDTO>> LoginAsync(LoginDTO dto)
    {
        var email = dto.Email.Trim().ToLowerInvariant();
        var user = await _db.Users.FirstOrDefaultAsync(u => u.Email == email);

        if (user is null || !BCrypt.Net.BCrypt.Verify(dto.Password, user.PasswordHash))
            return ServiceResult<AuthResponseDTO>.Fail("Email hoặc mật khẩu không đúng.", ServiceErrorType.Unauthorized);

        if (!user.IsActive)
            return ServiceResult<AuthResponseDTO>.Fail("Tài khoản đã bị khóa.", ServiceErrorType.Forbidden);

        return ServiceResult<AuthResponseDTO>.Ok(_tokenService.CreateToken(user));
    }
}

using BookStoreApi.Data;
using BookStoreApi.DTOs.Admin;
using BookStoreApi.DTOs.Catalog;
using BookStoreApi.Models;

namespace BookStoreApi.Services.Catalog;

public class ProductService : IProductService
{
    private readonly AppDbContext _db;
    private readonly IUploadService _uploadService;

    public ProductService(AppDbContext db, IUploadService uploadService)
    {
        _db = db;
        _uploadService = uploadService;
    }

    private IQueryable<Product> BaseQuery() =>
        _db.Products
            .Include(p => p.Category)
            .Include(p => p.BookDetail!).ThenInclude(bd => bd!.Publisher)
            .Include(p => p.BookDetail!).ThenInclude(bd => bd!.Authors);

    public async Task<ServiceResult<object>> SearchAsync(string? q, int? categoryId, int page, int pageSize, bool includeInactive = false)
    {
        var query = BaseQuery().AsNoTracking();

        if (!includeInactive)
            query = query.Where(p => p.IsActive);

        if (!string.IsNullOrWhiteSpace(q))
            query = query.Where(p => p.Name.Contains(q));

        if (categoryId is not null)
            query = query.Where(p => p.CategoryId == categoryId);

        var total = await query.CountAsync();
        var items = await query
            .OrderBy(p => p.Name)
            .Skip((page - 1) * pageSize)
            .Take(pageSize)
            .ToListAsync();   // lấy entity về trước, map DTO sau (ToDto không dịch được sang SQL)

        return ServiceResult<object>.Ok(new
        {
            total,
            page,
            pageSize,
            items = items.Select(p => p.ToDto()).ToList()
        });
    }

    public async Task<ServiceResult<ProductDTO>> GetByIdAsync(int id)
    {
        var product = await BaseQuery().AsNoTracking().FirstOrDefaultAsync(p => p.Id == id);
        if (product is null)
            return ServiceResult<ProductDTO>.Fail("Không tìm thấy sản phẩm.", ServiceErrorType.NotFound);

        return ServiceResult<ProductDTO>.Ok(product.ToDto());
    }

    public async Task<ServiceResult<ProductDTO>> CreateAsync(CreateProductDTO dto)
    {
        var category = await _db.Categories.FindAsync(dto.CategoryId);
        if (category is null)
            return ServiceResult<ProductDTO>.Fail("Danh mục không tồn tại.", ServiceErrorType.NotFound);

        var name = dto.Name.Trim();
        var slug = await GenerateUniqueSlugAsync(name);

        var product = new Product
        {
            Name = name,
            Slug = slug,
            Description = dto.Description,
            Price = dto.Price,
            Stock = dto.Stock,
            ImageUrls = dto.ImageUrls,
            CategoryId = dto.CategoryId
        };

        // Chỉ gắn BookDetail nếu Category thuộc nhóm Sách
        if (category.Type == CategoryType.Book && dto.BookDetail is not null)
        {
            var bookDetailResult = await BuildBookDetailAsync(dto.BookDetail);
            if (!bookDetailResult.Success)
                return ServiceResult<ProductDTO>.Fail(bookDetailResult.ErrorMessage!, bookDetailResult.ErrorType);

            product.BookDetail = bookDetailResult.Data;
        }

        _db.Products.Add(product);
        await _db.SaveChangesAsync();

        return await GetByIdAsync(product.Id);
    }

    public async Task<ServiceResult<ProductDTO>> UpdateAsync(int id, UpdateProductDTO dto)
    {
        var product = await _db.Products
            .Include(p => p.BookDetail)
            .ThenInclude(bd => bd.Authors) // Bắt buộc Include để EF track bảng trung gian
            .FirstOrDefaultAsync(p => p.Id == id);

        if (product is null)
            return ServiceResult<ProductDTO>.Fail("Không tìm thấy sản phẩm.", ServiceErrorType.NotFound);

        var category = await _db.Categories.FindAsync(dto.CategoryId);
        if (category is null)
            return ServiceResult<ProductDTO>.Fail("Danh mục không tồn tại.", ServiceErrorType.NotFound);

        var name = dto.Name.Trim();
        if (product.Name != name)
            product.Slug = await GenerateUniqueSlugAsync(name, excludeId: id);

        product.Name = name;
        product.Description = dto.Description;
        product.Price = dto.Price;
        product.Stock = dto.Stock;
        product.ImageUrls = dto.ImageUrls;
        product.CategoryId = dto.CategoryId;
        product.IsActive = dto.IsActive;
        product.UpdatedAt = DateTime.UtcNow;

        if (category.Type == CategoryType.Book && dto.BookDetail is not null)
        {
            var bookDetailResult = await BuildBookDetailAsync(dto.BookDetail, existing: product.BookDetail);
            if (!bookDetailResult.Success)
                return ServiceResult<ProductDTO>.Fail(bookDetailResult.ErrorMessage!, bookDetailResult.ErrorType);

            product.BookDetail = bookDetailResult.Data;
        }
        else if (category.Type != CategoryType.Book)
        {
            // Đổi Category sang nhóm "Khác" thì gỡ bỏ BookDetail cũ (nếu có)
            if (product.BookDetail is not null)
                _db.BookDetails.Remove(product.BookDetail);
        }

        // Lọc ra danh sách ảnh cũ bị loại bỏ khỏi danh sách ảnh mới
        var removedImages = product.ImageUrls
            .Except(dto.ImageUrls ?? new List<string>(), StringComparer.OrdinalIgnoreCase)
            .ToList();

        // Gán danh sách ảnh mới như bình thường
        product.ImageUrls = dto.ImageUrls ?? new List<string>();
        // ... (cập nhật các trường khác và BookDetail)

        await _db.SaveChangesAsync();

        // Chỉ xóa file vật lý SAU KHI đã lưu Database thành công!
        if (removedImages.Count > 0)
        {
            _uploadService.DeleteFiles(removedImages);
        }

        return await GetByIdAsync(product.Id);

        // await _db.SaveChangesAsync();
        // return await GetByIdAsync(product.Id);
    }

    {
        var product = await _db.Products.FindAsync(id);
        if (product is null)
            return ServiceResult<bool>.Fail("Không tìm thấy sản phẩm.", ServiceErrorType.NotFound);

        _db.Products.Remove(product);   // BookDetail tự xóa theo nhờ Cascade

        // Xóa toàn bộ ảnh vật lý của sách sau khi xóa khỏi DB

        return ServiceResult<bool>.Ok(true);
    }

    // Tạo hoặc cập nhật BookDetail dùng chung cho cả Create và Update
    private async Task<ServiceResult<BookDetail>> BuildBookDetailAsync(
        CreateBookDetailDTO dto, BookDetail? existing = null)
    {
        if (!string.IsNullOrWhiteSpace(dto.Isbn))
        {
            var isbnTaken = await _db.BookDetails.AnyAsync(bd =>
                bd.Isbn == dto.Isbn && (existing == null || bd.ProductId != existing.ProductId));

            if (isbnTaken)
                return ServiceResult<BookDetail>.Fail("ISBN đã tồn tại.", ServiceErrorType.Conflict);
        }

        var bookDetail = existing ?? new BookDetail();
        bookDetail.Isbn = dto.Isbn;
        bookDetail.Pages = dto.Pages;
        bookDetail.Language = dto.Language;
        bookDetail.PublishedDate = dto.PublishedDate;
        bookDetail.PublisherId = dto.PublisherId;

        if (dto.AuthorIds.Count > 0)
        {
            var authors = await _db.Authors
                .Where(a => dto.AuthorIds.Contains(a.Id))
                .ToListAsync();

            if (authors.Count != dto.AuthorIds.Count)
                return ServiceResult<BookDetail>.Fail("Có tác giả không tồn tại trong hệ thống.", ServiceErrorType.NotFound);

            bookDetail.Authors = authors;
        }
        else
        {
            bookDetail.Authors.Clear(); // Xóa toàn bộ tác giả nếu truyền mảng rỗng
        }

        return ServiceResult<BookDetail>.Ok(bookDetail);
    }

    private async Task<string> GenerateUniqueSlugAsync(string name, int? excludeId = null)
    {
        var baseSlug = name.ToSlug();
        var slug = baseSlug;
        var counter = 2;

        while (await _db.Products.AnyAsync(p => p.Slug == slug && p.Id != excludeId))
        {
            slug = $"{baseSlug}-{counter}";
            counter++;
        }

        return slug;
    }

namespace BookStoreApi.Services.Upload;

public class UploadService : IUploadService
{
    private readonly IWebHostEnvironment _env;
    private static readonly string[] AllowedExtensions = { ".jpg", ".jpeg", ".png", ".webp", ".gif" };
    private const long MaxFileSize = 5 * 1024 * 1024; // 5 MB
    private const int MaxFilesPerRequest = 10;

    public UploadService(IWebHostEnvironment env) => _env = env;

    public ServiceResult<UploadResultDTO> SaveImage(IFormFile file)
    {
        var result = SaveSingleFile(file);
        return result.Error is null
            ? ServiceResult<UploadResultDTO>.Ok(result.Data!)
            : ServiceResult<UploadResultDTO>.Fail(result.Error);
    }

    public UploadMultipleResultDTO SaveImages(List<IFormFile> files)
    {
        var response = new UploadMultipleResultDTO();

        if (files is null || files.Count == 0)
        {
            response.Errors.Add("Vui lòng chọn ít nhất một ảnh.");
            return response;
        }

        if (files.Count > MaxFilesPerRequest)
        {
            response.Errors.Add($"Chỉ được upload tối đa {MaxFilesPerRequest} ảnh mỗi lần.");
            return response;
        }

        foreach (var file in files)
        {
            var result = SaveSingleFile(file);
            if (result.Error is not null)
                response.Errors.Add($"{file.FileName}: {result.Error}");
            else
                response.Uploaded.Add(result.Data!);
        }

        return response;
    }

    // Hàm dùng chung cho cả single và multiple, tránh lặp code
    private (UploadResultDTO? Data, string? Error) SaveSingleFile(IFormFile file)
    {
        if (file is null || file.Length == 0)
            return (null, "File rỗng hoặc không hợp lệ.");

        if (file.Length > MaxFileSize)
            return (null, "Ảnh không được vượt quá 5MB.");

        var ext = Path.GetExtension(file.FileName).ToLowerInvariant();
        if (!AllowedExtensions.Contains(ext))
            return (null, "Chỉ chấp nhận file ảnh (jpg, png, webp, gif).");

        var fileName = $"{Guid.NewGuid()}{ext}";
        var uploadsFolder = Path.Combine(_env.WebRootPath ?? "wwwroot", "uploads");

        if (!Directory.Exists(uploadsFolder))
            Directory.CreateDirectory(uploadsFolder);

        var filePath = Path.Combine(uploadsFolder, fileName);

        using (var stream = new FileStream(filePath, FileMode.Create))
        {
            file.CopyTo(stream);
        }

        return (new UploadResultDTO { Url = $"/uploads/{fileName}", FileName = fileName }, null);
    }

    // Thêm vào cuối class UploadService
    public void DeleteFiles(IEnumerable<string> relativeUrls)
    {
        var uploadsFolder = Path.Combine(_env.WebRootPath ?? "wwwroot", "uploads");

        foreach (var url in relativeUrls)
        {
            // Chỉ xóa những file nội bộ bắt đầu bằng "/uploads/", bỏ qua link ảnh ngoài (https://...)
            if (string.IsNullOrWhiteSpace(url) || !url.StartsWith("/uploads/", StringComparison.OrdinalIgnoreCase))
                continue;

            // Lấy tên file an toàn (chống lỗi Path Traversal như "../../")
            var fileName = Path.GetFileName(url);
            var filePath = Path.Combine(uploadsFolder, fileName);

            if (File.Exists(filePath))
            {
                File.Delete(filePath);
            }
        }
    }
}

// Services/Users/UserService.cs
using BookStoreApi.Data;
using BookStoreApi.DTOs.User;

namespace BookStoreApi.Services.Users;

public class UserService : IUserService
{
    private readonly AppDbContext _db;

    public UserService(AppDbContext db) => _db = db;

    public async Task<ServiceResult<UserDTO>> GetMeAsync(int userId)
    {
        var user = await _db.Users.FindAsync(userId);
        if (user is null) return ServiceResult<UserDTO>.Fail("Không tìm thấy user.", ServiceErrorType.NotFound);
        return ServiceResult<UserDTO>.Ok(user.ToDTO());
    }

    public async Task<ServiceResult<UserDTO>> UpdateProfileAsync(int userId, UpdateProfileDTO dto)
    {
        var user = await _db.Users.FindAsync(userId);
        if (user is null) return ServiceResult<UserDTO>.Fail("Không tìm thấy user.", ServiceErrorType.NotFound);

        user.FullName = dto.FullName.Trim();
        user.PhoneNumber = dto.PhoneNumber ?? string.Empty;
        user.Address = dto.Address ?? string.Empty;
        user.AvatarUrl = dto.AvatarUrl;
        user.UpdatedAt = DateTime.UtcNow;

        await _db.SaveChangesAsync();
        return ServiceResult<UserDTO>.Ok(user.ToDTO());
    }

    public async Task<ServiceResult<bool>> ChangePasswordAsync(int userId, ChangePasswordDTO dto)
    {
        var user = await _db.Users.FindAsync(userId);
        if (user is null) return ServiceResult<bool>.Fail("Không tìm thấy user.", ServiceErrorType.NotFound);

        if (!BCrypt.Net.BCrypt.Verify(dto.CurrentPassword, user.PasswordHash))
            return ServiceResult<bool>.Fail("Mật khẩu hiện tại không đúng.");

        user.PasswordHash = BCrypt.Net.BCrypt.HashPassword(dto.NewPassword);
        user.UpdatedAt = DateTime.UtcNow;

        await _db.SaveChangesAsync();
        return ServiceResult<bool>.Ok(true);
    }
}
